--[[
    Glass UI - Complete Example
    Current distribution:
    https://raw.githubusercontent.com/nelsonbeiroa4-dotcom/Glass-UI/main/dist/GlassUI.lua
]]

local Library = loadstring(game:HttpGet(
    "https://raw.githubusercontent.com/nelsonbeiroa4-dotcom/Glass-UI/main/dist/GlassUI.lua"
))()

local Window = Library:CreateWindow({
    Title = "Glass UI | Example",
    Footer = "Glass UI • Example build",
    Size = UDim2.fromOffset(780, 560),
    AutoShow = true,
    Blur = true,
    BlurSize = 18,
    ToggleKeybind = Enum.KeyCode.RightControl,
})

-- MAIN
local Main = Window:AddTab("Main", "house")
local General = Main:AddLeftGroupbox("General")
local Controls = Main:AddRightGroupbox("Controls")

General:AddLabel({
    Text = "A compact glassmorphism UI built from a small, consistent API.",
    DoesWrap = true,
})

General:AddButton("Show notification", function()
    Library:Notify({
        Title = "Glass UI",
        Description = "The notification stack is working.",
        Time = 4,
    })
end)

General:AddToggle("ExampleToggle", {
    Text = "Enable example",
    Default = false,
    Callback = function(value)
        print("Example toggle:", value)
    end,
}):AddColorPicker("ExampleToggleColor", {
    Title = "Toggle color",
    Default = Color3.fromRGB(191, 91, 255),
})

General:AddToggle("DependentToggle", {
    Text = "Show advanced options",
    Default = false,
})

local Dependent = General:AddDependencyBox()
Dependent:SetupDependencies({
    { Library.Toggles.DependentToggle, true },
})
Dependent:AddSlider("AdvancedAmount", {
    Text = "Advanced amount",
    Min = 0,
    Max = 100,
    Default = 35,
    Rounding = 0,
})

-- INPUTS AND SELECTION
Controls:AddSlider("ExampleSlider", {
    Text = "Intensity",
    Min = 0,
    Max = 100,
    Default = 50,
    Suffix = "%",
    Rounding = 0,
    Callback = function(value)
        print("Slider:", value)
    end,
})

Controls:AddDropdown("Mode", {
    Text = "Select mode",
    Values = { "Legit", "Default", "Custom" },
    Default = "Default",
    Callback = function(value)
        print("Mode:", value)
    end,
})

Controls:AddDropdown("MultiModes", {
    Text = "Multiple modes",
    Values = { "A", "B", "C", "D" },
    Multi = true,
    Default = { "A" },
    Callback = function(values)
        print("Selected:", table.concat(values, ", "))
    end,
})

Controls:AddInput("PlayerName", {
    Text = "Text input",
    Placeholder = "Type something...",
    Default = "",
    Finished = true,
    Callback = function(value)
        print("Input:", value)
    end,
})

Controls:AddColorPicker("StandaloneColor", {
    Title = "Standalone color",
    Default = Color3.fromRGB(80, 150, 255),
})

Controls:AddKeyPicker("ExampleKey", {
    Text = "Example key",
    Default = "G",
    Mode = "Toggle",
    Callback = function(active)
        print("Key state:", active)
    end,
})

Controls:AddDivider("Media")
Controls:AddImage("ExampleImage", {
    Image = "rbxassetid://0",
    Height = 100,
})

-- VISUALS AND NESTED TABS
local Visuals = Window:AddTab("Visuals", "eye", "Media and visual elements")
local VisualGroup = Visuals:AddLeftGroupbox("Viewport")
VisualGroup:AddViewport("ExampleViewport", {
    Object = workspace:FindFirstChild("Part"),
    Clone = true,
    Height = 160,
})

local Sections = Visuals:AddRightTabbox("Sections")
local First = Sections:AddTab("First")
First:AddToggle("SubToggle", { Text = "Sub-tab toggle", Default = false })

local Second = Sections:AddTab("Second")
Second:AddButton("Second page button", function()
    Library:Notify("This button belongs to the second sub-tab.", 3)
end)

-- APPEARANCE AND CONFIGURATIONS
local Settings = Window:AddTab("Settings", "settings")
Library.ThemeManager:ApplyToTab(Settings)
Library.SaveManager:BuildConfigSection(Settings)

local Watermark = Library:CreateWatermark({
    Text = "Glass UI • Example",
    Visible = true,
    Position = UDim2.fromOffset(14, 14),
})

Library:Notify({
    Title = "Loaded",
    Description = "Glass UI is ready.",
    Time = 3,
})
