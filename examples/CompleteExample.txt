local Root = "https://raw.githubusercontent.com/nelsonbeiroa4-dotcom/Glass-UI/main/"

local function LoadModule(Path)
    local Source = game:HttpGet(Root .. Path)
    local Chunk, CompileError = loadstring(Source)
    assert(Chunk, CompileError or ("Could not compile " .. Path))
    return Chunk()
end

local Library = LoadModule("dist/GlassUI.lua")
local ThemeManager = LoadModule("addons/ThemeManager.lua")
local SaveManager = LoadModule("addons/SaveManager.lua")

ThemeManager:SetLibrary(Library)
SaveManager:SetLibrary(Library)
ThemeManager:SetFolder("GlassUI")
SaveManager:SetFolder("GlassUI")

print("[Glass UI] Version:", Library.Version)
print("[Glass UI] Executor:", Library.Compatibility.ExecutorName)
print("[Glass UI] Recognized:", Library.Compatibility.IsSupported)

local Window = Library:CreateWindow({
    Title = "Glass UI",
    Footer = "Glass UI v0.3.1  •  RightControl to hide/show",
    Size = UDim2.fromOffset(960, 620),
    Center = true,
    AutoShow = true,
    Resizable = true,
    ShowCustomCursor = true,
    ToggleKeybind = Enum.KeyCode.RightControl,
})

local Tabs = {
    Home = Window:AddTab("Home", "house"),
    Visuals = Window:AddTab("Visuals", "eye"),
    Settings = Window:AddTab("Settings", "settings"),
}

local Welcome = Tabs.Home:AddGroupbox({
    Side = "Left",
    Name = "Welcome",
    Description = "Obsidian controls with the Glass UI visual theme.",
    IconName = "layout-dashboard",
})
Welcome:AddLabel("Glass UI is running. Drag the title bar to move the window and use RightControl to hide or show it.", true)
Welcome:AddButton({
    Text = "Test notification",
    Func = function()
        Library:Notify("Glass UI is working correctly.", 3)
    end,
})

local Status = Tabs.Home:AddGroupbox({
    Side = "Right",
    Name = "Status",
    Description = "Example toggles",
    IconName = "activity",
})
Status:AddToggle("ExampleToggle", {
    Text = "Enable example",
    Default = false,
    Callback = function(Value)
        print("Example toggle:", Value)
    end,
})
Status:AddToggle("SecondToggle", {
    Text = "Enable second option",
    Default = true,
    Callback = function(Value)
        print("Second option:", Value)
    end,
})

local Controls = Tabs.Visuals:AddGroupbox({
    Side = "Left",
    Name = "Controls",
    IconName = "sliders-horizontal",
})
Controls:AddSlider("Intensity", {
    Text = "Intensity",
    Default = 50,
    Min = 0,
    Max = 100,
    Rounding = 0,
    Suffix = "%",
    Callback = function(Value)
        print("Intensity:", Value)
    end,
})
Controls:AddDropdown("Mode", {
    Text = "Mode",
    Values = { "Default", "Soft", "Strong" },
    Default = "Default",
    Callback = function(Value)
        print("Mode:", Value)
    end,
})

local Inputs = Tabs.Visuals:AddGroupbox({
    Side = "Right",
    Name = "Inputs",
    IconName = "list-filter",
})
Inputs:AddInput("PlayerName", {
    Text = "Player name",
    Placeholder = "Type a name...",
    Default = "",
    Callback = function(Value)
        print("Name:", Value)
    end,
})
local AccentToggle = Inputs:AddToggle("UseAccent", {
    Text = "Enable accent preview",
    Default = true,
})
AccentToggle:AddColorPicker("AccentColor", {
    Title = "Accent color",
    Default = Color3.fromRGB(168, 85, 247),
    Callback = function(Value)
        print("Accent color:", Value)
    end,
})
Inputs:AddKeyPicker("ExampleKey", {
    Text = "Toggle key",
    Default = "G",
    Mode = "Toggle",
    Callback = function(Value)
        print("Key state:", Value)
    end,
})

ThemeManager:ApplyToTab(Tabs.Settings)
SaveManager:BuildConfigSection(Tabs.Settings)
Library:Notify("Glass UI v0.3.1 loaded.", 4)

-- Library:Unload()
